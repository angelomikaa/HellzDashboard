my_identity() contract (§B of the migration; house style = my_card 0013:281-303):
```sql
create or replace function public.my_identity()
returns jsonb
language plpgsql
stable
security definer
set search_path = public
as $$
declare
  v_member uuid := (auth.jwt()->>'member_id')::uuid;
begin
  if v_member is null
     or not exists (select 1 from members m where m.id = v_member and m.active) then
    raise exception 'forbidden';
  end if;
  return jsonb_build_object(
    'display_name', public.identity_display_name(v_member),
    'char_id', public.identity_char_id(v_member),
    'char_class', coalesce(
      (select h.char_class from member_identity_history h
        where h.member_id = v_member and h.char_class is not null
        order by h.created_at desc, h.id desc limit 1),
      (select m.char_class from members m where m.id = v_member)),
    'char_function', (select m.char_function from members m where m.id = v_member)
  );
end;
$$;

revoke execute on function public.my_identity() from public, anon;
grant  execute on function public.my_identity() to authenticated;
```
Semantics: display_name = newest history nickname with registry fallback (0014 identity_display_name); char_id = the same coalesce (identity_char_id) — safe to expose to the caller because the member types their own char_id on every submission and the RPC is caller-pinned; char_class = newest NON-NULL history class with registry fallback (null when both are null — SubmitPage's no-character blocked state); char_function = the registry pointer (staff-registered or member-promoted after 08-09; consumers read the pointer, never per-submission values).
