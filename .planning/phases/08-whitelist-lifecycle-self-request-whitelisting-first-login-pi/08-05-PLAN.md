export interface StaffMember extends MemberRow {
  char_id: string | null
  pin_set: boolean
  has_pending_proof: boolean
  identity: { char_id: string; nickname: string; char_class: string | null; char_function: 'dps'|'mdps'|'suporte'|null } | null
}
export interface IdentityQueueRow { proof_id: string; member_id: string; discord_handle: string; display_name: string;
  char_id: string; nickname: string; char_class: string | null; proof_path: string; created_at: string }
export interface IdentityHistoryRow { id: number; created_at: string; char_id: string; nickname: string;
  char_class: string | null; char_function: 'dps'|'mdps'|'suporte'|null; source: 'submission'|'correction'|'claim'|'request'; created_by: string | null }
export interface StaffRequestRow { id: string; request_type: 'member'|'prospect'; discord_handle: string;
  char_id: string; nickname: string; char_class: string; char_function: 'dps'|'mdps'|'suporte';
  proof_path: string; created_at: string; ovr: number | null; data_sufficient: boolean }
export function fetchStaffMembers(session): Promise<StaffMember[]>
export function addMember(session, handle, displayName): Promise<string>
export function resetMemberPin(session, memberId): Promise<void>
export function fetchIdentityQueue(session): Promise<IdentityQueueRow[]>
export function reviewIdentity(session, proofId, confirm: boolean): Promise<void>

export function approveRequest(session, requestId, charFunction): Promise<string>
export function rejectRequest(session, requestId): Promise<void>

export function identityProofUrl(session, path): string
export function deleteIdentityProof(session, path): Promise<void>

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| staff browser → admin RPCs | Every lifecycle action crosses a gated definer function |
| owner-only PIN reset | UI gating is UX; RPC re-checks role + owner active |
| storage proof preview | Staff-only authenticated object route |
| identity fields → rank/queue | Server-computed payloads; client renders verbatim |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-31 | Elevation of Privilege | reset PIN / role actions | mitigate | Owner-only UI + owner gate inside `reset_member_pin`; no client-side write path |
| T-08-32 | Information Disclosure | PIN state | mitigate | Only `pin_set` boolean is rendered; hash never selected/granted |
| T-08-33 | Information Disclosure | proof images | mitigate | Staff-only storage policy; authenticated object URL; no public bucket |
| T-08-34 | Tampering | approval | mitigate | Approve RPC re-checks pending + uniqueness + function whitelist; UI choice is not authoritative |
| T-08-35 | Repudiation | review decisions | mitigate | Server writes decided_at/by and review timestamps |
| T-08-36 | Spoofing | identity correction | mitigate | Staff-gated, validated formats, duplicate assertion, append-only history |
| T-08-37 | Denial of Service | double-click actions | mitigate | Single-flight busy lock; pending-only RPC scope surfaces not_pending |
</threat_model>
