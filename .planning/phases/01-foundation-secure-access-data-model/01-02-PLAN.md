```sql
-- NeoFoot Phase 1 RLS: every table locked; members see own rows only; append-only submissions
alter table members enable row level security;
alter table stat_catalog enable row level security;
alter table submissions enable row level security;
alter table submission_stats enable row level security;
alter table class_key_stats enable row level security;
alter table pin_login_attempts enable row level security;

-- members: read own row only; no INSERT/UPDATE/DELETE for authenticated
-- (registration = owner via service role / seed; role changes never via API)
create policy member_select_own on members
  for select to authenticated
  using (id::text = auth.jwt()->>'member_id');

-- submissions: select + insert own rows only; NO update/delete policies (append-only)
create policy member_select_own_submissions on submissions
  for select to authenticated
  using (member_id::text = auth.jwt()->>'member_id');

create policy member_insert_own_submissions on submissions
  for insert to authenticated
  with check (member_id::text = auth.jwt()->>'member_id');

-- submission_stats: select/insert via ownership of the parent submission; NO update/delete
create policy member_select_own_submission_stats on submission_stats
  for select to authenticated
  using (exists (
    select 1 from submissions s
    where s.id = submission_id and s.member_id::text = auth.jwt()->>'member_id'
  ));

create policy member_insert_own_submission_stats on submission_stats
  for insert to authenticated
  with check (exists (
    select 1 from submissions s
    where s.id = submission_id and s.member_id::text = auth.jwt()->>'member_id'
  ));

-- stat_catalog: readable by all authenticated (submit wizard needs it); writes via service role only
create policy authenticated_read_catalog on stat_catalog
  for select to authenticated
  using (true);

-- class_key_stats: NO policies = deny-all for every client credential (RNK-02 foundation;
-- key-stat config never crosses the API boundary to members)
-- pin_login_attempts: NO policies = service-role-only (written exclusively by verify-pin)
```

   ```typescript
   export interface LockoutState { failedCount: number; lockedUntil: Date | null }
   export const MAX_ATTEMPTS = 5
   export const LOCKOUT_MINUTES = 15

   export function isLocked(state: LockoutState, now: Date): boolean {
     return state.lockedUntil !== null && state.lockedUntil.getTime() > now.getTime()
   }
   export function retryAfterSeconds(state: LockoutState, now: Date): number {
     if (!isLocked(state, now)) return 0
     return Math.ceil((state.lockedUntil!.getTime() - now.getTime()) / 1000)
   }
   export function nextFailureState(state: LockoutState, now: Date): LockoutState {
     const failedCount = state.failedCount + 1
     const lockedUntil = failedCount >= MAX_ATTEMPTS ? new Date(now.getTime() + LOCKOUT_MINUTES * 60_000) : null
     return { failedCount, lockedUntil }
   }
   export function resetState(): LockoutState {
     return { failedCount: 0, lockedUntil: null }
   }
   ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| SPA -> verify-pin | Untrusted {discord_handle, pin} input crosses here |
| JWT -> PostgREST/RLS | Signed claims cross the data-access boundary |
| Edge Function -> DB | Service-role access (bypasses RLS) — must only serve the PIN gate |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Spoofing | JWT forgery | mitigate | HS256 signed with SUPABASE_JWT_SECRET (server-only env); 7-day expiry (D-06); role claim fixed to 'authenticated' |
| T-01-02 | Brute force | PIN attempts | mitigate | DB-backed lockout 5 -> 15 min per handle (D-07); attempts counter reset only on success; enforced in the Edge Function, never client-side |
| T-01-03 | Information disclosure | RLS bypass | mitigate | Policies on every table; identity from `auth.jwt()->>'member_id'` (server-verified claim, not client-supplied); class_key_stats deny-all |
| T-01-04 | Tampering | submissions mutation | mitigate | No UPDATE/DELETE policies for authenticated — append-only at the server (AUTH-02 / INPT-04 foundation) |
| T-01-05 | Elevation of privilege | role claims | mitigate | member_role read from the members row at login (server truth), never accepted from client input |
| T-01-06 | DoS | verify-pin flood | accept | Free-tier + per-handle lockout bounds damage; low-value target |
| T-01-07 | Information disclosure | user enumeration | mitigate | Missing handle and wrong PIN return identical 401 responses |
</threat_model>
