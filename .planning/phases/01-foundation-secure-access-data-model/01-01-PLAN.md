   ```sql
   -- NeoFoot Phase 1 schema: members, catalog, append-only EAV submissions, key-stat config, PIN lockout
   create table members (
     id uuid primary key default gen_random_uuid(),
     discord_handle text not null unique,          -- D-01/D-04: identity key, unique at DB level
     display_name text not null,                   -- in-game nickname
     pin_hash text not null,                       -- D-03: bcrypt hash, never plaintext
     role text not null default 'member' check (role in ('member','officer','owner')),
     active boolean not null default true,         -- Phase 6 removal revokes access via active=false
     created_at timestamptz not null default now()
   );

   create table stat_catalog (
     stat_key text primary key,                    -- e.g. 'crit'
     label text not null,                          -- e.g. 'CRIT'
     group_name text not null check (group_name in ('general','quasi','special')),
     value_type text not null check (value_type in ('flat','percentual'))
   );

   create table submissions (
     id uuid primary key default gen_random_uuid(),
     member_id uuid not null references members(id),
     status text not null default 'pending' check (status in ('pending','validated','rejected')),
     submitted_at timestamptz not null default now()
   );

   create table submission_stats (
     id bigserial primary key,
     submission_id uuid not null references submissions(id) on delete cascade,
     stat_key text not null references stat_catalog(stat_key),
     value numeric not null,
     unique (submission_id, stat_key)              -- EAV: one row per stat per submission
   );

   create table class_key_stats (
     class_name text not null,
     stat_key text not null references stat_catalog(stat_key),
     primary key (class_name, stat_key)            -- Phase 2/3 populate; exactly-6-per-class enforced in Phase 3 logic
   );

   create table pin_login_attempts (
     discord_handle text primary key,
     failed_count int not null default 0,
     locked_until timestamptz,
     updated_at timestamptz not null default now()
   );

   create index idx_submissions_member on submissions(member_id);
   create index idx_submission_stats_submission on submission_stats(submission_id);
   ```
   Note: `class_name` is plain text (no classes table) — one member = one character (D-02, 1:1); Phase 2's extensible class list (REG-02) decides whether a lookup table is added.

   ```sql
   -- and replaces the example handles/hashes below. Re-runnable via upsert.

     ('owner-handle', 'Owner Nickname', '<bcrypt hash from scripts/hash-pin.mjs>', 'owner', true),

   ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client bundle -> env | Only VITE_* vars may reach the client; service_role/JWT secret must never cross |
| SPA -> PostgREST | All data access crosses RLS here (policies arrive in Plan 01-02) |
| seed script -> members table | PIN hashes enter the DB; plaintext PINs never do |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Spoofing | JWT trust | mitigate | JWTs signed HS256 with `SUPABASE_JWT_SECRET` (Plan 01-02); secret never in client bundle or .env.example |
| T-01-02 | Brute force | PIN auth | mitigate | Server-side lockout 5 wrong -> 15 min (D-07, Plan 01-02); `pin_login_attempts` table created here |
| T-01-03 | Information disclosure | Anon/data reads | mitigate | RLS on every table (Plan 01-02); `class_key_stats` deny-all; schema has no public-by-default tables |
| T-01-04 | Tampering | submissions/submission_stats | mitigate | Append-only enforced via RLS (no UPDATE/DELETE policies, Plan 01-02); EAV unique constraint prevents duplicate stat rows |
| T-01-05 | Elevation of privilege | members.role | mitigate | `members` not writable via API (no INSERT/UPDATE policies for authenticated, Plan 01-02); role changes only via service role / owner seed |
| T-01-06 | DoS | verify-pin / free tier | accept | Free-tier target with per-handle lockout limiting damage; Supabase absorbs invocation floods up to 500k/month |
</threat_model>
