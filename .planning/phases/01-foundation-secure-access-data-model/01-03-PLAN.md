   ```typescript
   export interface NeoFootSession {
     access_token: string
     member_id: string
     display_name: string
     member_role: 'member' | 'officer' | 'owner'
     expires_at: number // epoch seconds
   }
   const STORAGE_KEY = 'nefoot_session'
   export function saveSession(s: NeoFootSession): void { localStorage.setItem(STORAGE_KEY, JSON.stringify(s)) }
   export function loadSession(): NeoFootSession | null {
     const raw = localStorage.getItem(STORAGE_KEY)
     return raw ? JSON.parse(raw) as NeoFootSession : null
   }
   export function clearSession(): void { localStorage.removeItem(STORAGE_KEY) }
   export function isExpired(s: NeoFootSession): boolean { return s.expires_at * 1000 <= Date.now() }
   ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| LoginPage -> verify-pin | Untrusted credentials cross here |
| localStorage <-> session helpers | Client-side state — UX only, never the enforcement layer |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Spoofing | Token in localStorage | mitigate | JWT is server-signed with 7-day expiry (Plan 01-02); RLS re-validates every request server-side — a stolen token only works while valid and grants only its own rows |
| T-01-02 | Brute force | Client-side PIN entry | mitigate | Lockout is server-side (D-07) — client rate limiting deliberately absent so it cannot be bypassed |
| T-01-03 | Information disclosure | Plaintext PIN in storage/logs | mitigate | PIN sent only in the request body over HTTPS; never stored, never logged; `type="password"` input |
| T-01-08 | Tampering | Client-side guard bypass | accept | Route guards are UX only by design (ARCHITECTURE.md) — real enforcement is RLS, proven in Plan 01-02 |
</threat_model>
