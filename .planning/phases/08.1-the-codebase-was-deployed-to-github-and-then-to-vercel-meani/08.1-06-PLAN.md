    ```sql
    -- 0023_sentinel_identity_guard.sql
    -- 08.1-06 (D-11, D-16 replacement): retire the placeholder identity namespace.
    --
    -- D-16 proposed a separate `stf-`/`off-` prefix for staff handles. The live
    -- registry has 8 officers + 1 owner on REAL Discord handles, so renaming was
    -- rejected as impractical. What is actually dangerous is the inverse: a
    -- placeholder-shaped handle (`*-handle`) that carries staff privilege or a
    -- login-capable PIN. This migration closes that namespace in the database,
    -- not in the UI.
    --
    -- Apply to LIVE: node scripts/apply-seed.mjs supabase/migrations/0023_sentinel_identity_guard.sql

    begin;

    -- 1. Neutralize every sentinel-shaped row: ordinary member, inactive, no PIN.
    update public.members
       set role = 'member',
           active = false,
           pin_hash = null,
           claimed_at = null
     where discord_handle ~* '-handle$'
       and (role in ('officer', 'owner') or pin_hash is not null or active);

    -- 2. Clear lockout state for those identities (0020 keys attempts by member id,
    --    pre-0020 rows used the handle — clear both shapes).
    delete from public.pin_login_attempts
     where discord_handle in (
       select m.id::text from public.members m where m.discord_handle ~* '-handle$'
       union
       select m.discord_handle from public.members m where m.discord_handle ~* '-handle$'
     );

    -- 3. Hard guard: a sentinel-shaped handle can never be staff and never hold a
    --    PIN. Evaluated on every insert/update, so add_member, setup-pin's claim and
    --    a raw Management API write are all covered.
    alter table public.members
      drop constraint if exists members_sentinel_not_privileged;
    alter table public.members
      add constraint members_sentinel_not_privileged
      check (
        not (
          discord_handle ~* '-handle$'
          and (role in ('officer', 'owner') or pin_hash is not null)
        )
      );

    comment on constraint members_sentinel_not_privileged on public.members is
      'Placeholder-shaped handles (*-handle) are inert by construction: never officer/owner, never PIN-capable. Replaces the rejected stf-/off- staff-prefix namespace (08.1 D-16, migration 0023).';

    commit;
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Repo history -> attacker | Anyone who ever obtains a clone reads the committed owner hash/handle |
| Placeholder namespace -> privilege | A `*-handle` row that could hold staff role or a PIN is a hijack surface |
| Owner terminal -> live DB | The rotation tool holds a Management API token and writes `pin_hash` |
| Seed file -> live DB | A re-run of a seed is an unauthenticated write path to `members` |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-081-06-01 | Spoofing | committed owner bcrypt hash | mitigate | PIN rotated out-of-band (task 3) + repo gate (08.1-09) + seed scrub (task 2) |
| T-081-06-02 | Elevation of Privilege | `officer-handle` / `member-handle` rows | mitigate | Migration 0023 neutralizes them and adds `members_sentinel_not_privileged` (checked on every write, including Management API writes) |
| T-081-06-03 | Elevation of Privilege | re-creating a sentinel staff row | mitigate | CHECK constraint rejects `*-handle` + officer/owner or non-null `pin_hash` with `23514` |
| T-081-06-04 | Tampering | re-running `whitelist.sql` | mitigate | Seed contains only inert placeholders and `do update set display_name` — it cannot overwrite a live `pin_hash`, `role` or `active` |
| T-081-06-05 | Information Disclosure | PIN echoed by the rotation tool | mitigate | Tool prints only handle/role/hash length; PIN read with echo suppressed; no file write |
| T-081-06-06 | Denial of Service | constraint blocking a legitimate member whose Discord handle ends in `-handle` | accept | Pre-flight query (task 1 step 0) must show only placeholder rows; if a real handle appears, the plan blocks instead of applying |
| T-081-06-07 | Repudiation | history still contains the old hash | accept | D-11 explicit decision: rotate, accept history, no rewrite. Old hash is invalid after task 3 |
