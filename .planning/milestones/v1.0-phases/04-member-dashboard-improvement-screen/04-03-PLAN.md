Fetch — mount-only, exactly ONE call, cancelled-flag cleanup (KeyStatsPage pattern):
```typescript
useEffect(() => {
  let cancelled = false
  fetchMyReport(session)
    .then((r) => { if (!cancelled) setReport(r) })
    .catch((err: unknown) => {
      if (cancelled) return
      setError(err instanceof ApiError && err.kind === 'unauthorized' ? SESSION_ERROR : NETWORK_ERROR)
    })
  return () => { cancelled = true }
}, [session])
```
No polling, no caching, no refresh button — switching tabs unmounts the page and returning refetches (compute-on-read; that IS the refresh mechanism). The page's only data import from lib/api is `fetchMyReport` + `ApiError` + `MyReportStat`/`MyReport` TYPES.

Handlers:
```typescript
function toggleGroup(g: GroupKey) {
  setOpenGroups((prev) => ({ ...prev, [g]: !prev[g] }))
}
function handleFocusJump(statKey: string, group: GroupKey) {
  setOpenGroups((prev) => ({ ...prev, [group]: true }))   // parent owns state so jumps can expand
  requestAnimationFrame(() => {
    const el = document.getElementById(`stat-${statKey}`)
    el?.scrollIntoView({ behavior: 'smooth', block: 'center' })
    el?.focus()   // tabIndex={-1} is on the StatCard wrapper — programmatic focus only
  })
}
```
